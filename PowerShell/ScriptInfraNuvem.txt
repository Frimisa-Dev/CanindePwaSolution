<#
.SYNOPSIS
  ScriptInfra.ps1 - Configuração oficial do IIS / AppPool para CurioWebServer
.DESCRIPTION
  - Cria o AppPool CurioWebServerPool se não existir
  - Define runtime, identidade e variáveis sensíveis no AppPool
  - Não utiliza variáveis de máquina (/M), isolando o backend
  - Script determinístico e reexecutável
#>

Import-Module WebAdministration

$POOL_NAME = "CurioWebServerPool"
$APPCMD    = "$env:SystemRoot\System32\inetsrv\appcmd.exe"

Write-Host "=== CURIO | CONFIGURAÇÃO IIS / APPPOOL ===" -ForegroundColor Cyan

# -------------------------------------------------------------------------------------
# 1. GARANTIA DO APPPOOL
# -------------------------------------------------------------------------------------
if (-not (Test-Path "IIS:\AppPools\$POOL_NAME")) {
    New-WebAppPool $POOL_NAME
    Write-Host "✔ AppPool criado: $POOL_NAME" -ForegroundColor Green
}

Set-ItemProperty "IIS:\AppPools\$POOL_NAME" -Name managedRuntimeVersion -Value ""
Set-ItemProperty "IIS:\AppPools\$POOL_NAME" -Name processModel.identityType -Value ApplicationPoolIdentity

# -------------------------------------------------------------------------------------
# 2. LIMPEZA DE VARIÁVEIS EXISTENTES
# -------------------------------------------------------------------------------------
& $APPCMD set config /section:applicationPools "/[name='$POOL_NAME'].processModel.environmentVariables" /commit:apphost
Write-Host "✔ Variáveis antigas removidas" -ForegroundColor Yellow

# -------------------------------------------------------------------------------------
# 3. DEFINIÇÃO DAS VARIÁVEIS SENSÍVEIS
# -------------------------------------------------------------------------------------
$envVars = @{
    "CURIO__ENVIRONMENT" = "PRODUCTION"
    "CURIO__
